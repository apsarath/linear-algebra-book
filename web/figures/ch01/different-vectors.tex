\begin{tikzpicture}[>=Stealth, thick, scale=0.8,
    u/.style={->, aopsblue, line width=1.2pt},
    v/.style={->, aopsred, line width=1.2pt},
    ghost/.style={->, aopsred!55, dashed, line width=1pt},
    paneltitle/.style={font=\small\bfseries, text=black!75}]

  % (a) same direction, different length
  \begin{scope}
    \draw[step=1, black!10, thin] (-1.5,-2) grid (4.5,3);

  \end{scope}
